library(shiny)

# Enhancement: sliderInput to filter by cylinder count.
# Users can check if the predictor-mpg relationship holds within 4-, 6-, or 8-cylinder subgroups.
# This reveals whether engine size confounds the association between the selected predictor and mpg.

ui <- fluidPage(
  titlePanel("mtcars Explorer"),
  sidebarLayout(
    sidebarPanel(
      selectInput("predictor", "Predictor Variable",
                  choices = c("wt", "hp", "disp", "drat")),
      sliderInput("cyl_filter", "Number of Cylinders",
                  min = 4, max = 8, value = c(4, 8), step = 2)
    ),
    mainPanel(
      plotOutput("scatter"),
      verbatimTextOutput("model_summary")
    )
  )
)

server <- function(input, output) {

  filtered_data <- reactive({
    mtcars[mtcars$cyl >= input$cyl_filter[1] &
           mtcars$cyl <= input$cyl_filter[2], ]
  })

  fitted_model <- reactive({
    lm(as.formula(paste("mpg ~", input$predictor)),
       data = filtered_data())
  })

  output$scatter <- renderPlot({
    df <- filtered_data()
    ggplot(df, aes(x = .data[[input$predictor]], y = mpg)) +
      geom_point(size = 2.5, alpha = 0.7) +
      geom_smooth(method = "lm", se = TRUE, color = "#0072B2") +
      labs(title = paste("mpg vs", input$predictor),
           x = input$predictor, y = "mpg") +
      theme_minimal()
  })

  output$model_summary <- renderPrint({
    summary(fitted_model())
  })
}

shinyApp(ui, server)

#(a)
bootstrap_ci_sequential <- function(data, n_bootstrap = 1000, confidence = 0.95) {
  stats <- replicate(n_bootstrap, mean(sample(data, replace = TRUE)))
  alpha <- 1 - confidence
  setNames(quantile(stats, c(alpha / 2, 1 - alpha / 2)), c("lower", "upper"))
}

#(b)
bootstrap_ci_parallel <- function(data, n_bootstrap = 1000, confidence = 0.95) {
  cl <- makeCluster(safe_cores)
  stats <- parSapply(cl, seq_len(n_bootstrap),
                     function(i) mean(sample(data, replace = TRUE)))
  stopCluster(cl)
  alpha <- 1 - confidence
  setNames(quantile(stats, c(alpha / 2, 1 - alpha / 2)), c("lower", "upper"))
}

#(c)
results <- lapply(c(500, 1000, 2000), function(n) {
  t_seq <- system.time(
    bootstrap_ci_sequential(mtcars$mpg, n_bootstrap = n)
  )["elapsed"]
  t_par <- system.time(
    bootstrap_ci_parallel(mtcars$mpg, n_bootstrap = n)
  )["elapsed"]
  data.frame(n_bootstrap  = n,
             sequential_s = round(t_seq, 3),
             parallel_s   = round(t_par, 3),
             speedup      = round(t_seq / t_par, 2))
})

summary_df <- do.call(rbind, results)
